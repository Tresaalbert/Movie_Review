const GOOGLE_SCRIPT_URL = "PASTE_YOUR_WEB_APP_URL_HERE"; document.getElementById("reviewForm").addEventListener( "submit", async function(event) { event.preventDefault(); const button = this.querySelector(".submit"); const message = document.getElementById("success"); const review = { username: document.getElementById("username").value.trim(), district: document.getElementById("district").value.trim(), gender: document.getElementById("gender").value, movie: document.getElementById("movie").value.trim(), rating: document.getElementById("rating").value, comment: document.getElementById("comment").value.trim() }; if (!GOOGLE_SCRIPT_URL.startsWith("https://script.google.com/")) { message.textContent = "Please configure the Google Sheet URL."; return; } button.disabled = true; button.textContent = "Submitting..."; try { await fetch(GOOGLE_SCRIPT_URL, { method: "POST", mode: "no-cors", headers: { "Content-Type": "text/plain;charset=utf-8" }, body: JSON.stringify(review) }); message.textContent = "Your review has been sent. Please check the Google Sheet."; this.reset(); } catch (error) { message.textContent = "Unable to send review. Please try again."; } finally { button.disabled = false; button.textContent = "Submit Review"; } });
